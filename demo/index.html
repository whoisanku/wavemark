      Usage

      One import. No build step.

      ESM with bundled types, from a bundler or straight off a CDN.

      // npm i wavemark
import { wavemark } from 'wavemark';

wavemark('ankit', canvas);                       // draw into a canvas
wavemark('ankit', canvas, { style: 'pen' });    // or pick a style
wavemark('ankit', canvas, { breathe: true });   // slow in-place motion
img.src = wavemark.toDataURL('ankit', 64);       // PNG data URL
      <script type="module">
  import { wavemark } from 'https://cdn.jsdelivr.net/npm/wavemark/+esm';
  wavemark('ankit', document.querySelector('canvas'));
</script>
